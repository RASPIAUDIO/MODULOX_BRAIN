cmake_minimum_required(VERSION 3.19)
set(IDF_TARGET esp32s3)
set(MODULOX_MODEL "s3-16m8m" CACHE STRING "Modulox hardware profile")
file(READ "${CMAKE_CURRENT_LIST_DIR}/boards/profiles.json" BOARD_PROFILES)
foreach(KEY model project layout flashBytes psramBytes slotBytes ota0 ota1 sampleBytes)
    string(JSON BOARD_${KEY} GET "${BOARD_PROFILES}" "${MODULOX_MODEL}" "${KEY}")
endforeach()
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;boards/${MODULOX_MODEL}.defaults")
configure_file(boards/profile_config.h.in generated/board_profile_config.h @ONLY)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/version.txt" PROJECT_VER LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_LIST_DIR}/version.txt" "${CMAKE_CURRENT_LIST_DIR}/boards/profiles.json")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(COMPONENTS main)
project(${BOARD_project})
math(EXPR BOARD_FLASH_MB "${BOARD_flashBytes} / 1048576")
if(NOT CONFIG_ESPTOOLPY_FLASHSIZE STREQUAL "${BOARD_FLASH_MB}MB")
    message(FATAL_ERROR "sdkconfig does not match ${MODULOX_MODEL}: use its isolated build directory and sdkconfig")
endif()
if(MODULOX_MODEL STREQUAL "s3-32m16m")
    if(NOT CONFIG_ESPTOOLPY_OCT_FLASH OR NOT CONFIG_SPIFFS_PAGE_SIZE EQUAL 512)
        message(FATAL_ERROR "32/16 board requires octal flash and 512-byte SPIFFS pages")
    endif()
elseif(CONFIG_ESPTOOLPY_OCT_FLASH)
    message(FATAL_ERROR "16/8 board requires the original quad flash settings")
endif()
